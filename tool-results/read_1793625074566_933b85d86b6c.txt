     3→/* Engine generator soal Matematika otomatis per jenjang (SD/SMP/SMA),
     4→   kelas & topik. Semua soal dihitung ulang setiap kali dibuat, jawaban
     5→   selalu bilangan bulat / pecahan sederhana agar rapi untuk lembar kerja. */
     6→
     7→export interface GenSoal {
     8→  q: string;
     9→  a: string;
    10→}
    11→
    12→export interface TopikDef {
    13→  id: string;
    14→  label: string;
    15→  gen: (lv: number, R: () => number) => GenSoal;
    16→}
    17→
    18→/** bilangan bulat acak a..s.d. b (inklusif) */
    19→const ri = (R: () => number, a: number, b: number): number =>
    20→  a + Math.floor(R() * (b - a + 1));
    21→const pick = <T,>(R: () => number, arr: T[]): T => arr[Math.floor(R() * arr.length)];
    22→/** skala rentang berdasarkan tingkat kesulitan: 1=mudah 2=sedang 3=sulit */
    23→const S = (lv: number, base: number, inc: number) => Math.round(base * (1 + (lv - 1) * inc));
    24→
    25→const GCD = (a: number, b: number): number => (b === 0 ? Math.abs(a) : GCD(b, a % b));
    26→const LCM = (a: number, b: number): number => (a * b) / GCD(a, b);
    27→const simp = (p: number, q: number): string => {
    28→  const g = GCD(p, q) || 1;
    29→  let pp = p / g, qq = q / g;
    30→  if (qq < 0) { pp = -pp; qq = -qq; }
    31→  return qq === 1 ? `${pp}` : `${pp}/${qq}`;
    32→};
    33→/** desimal gaya Indonesia: 3,5 */
    34→const idn = (n: number): string => n.toString().replace(".", ",");

   859→/** Buat n soal unik dari satu topik pada mapel generator tertentu. */
   860→export function generateQuestions(
   861→  mapel: GenMapel,
   862→  topikId: string,
   863→  jenjang: string,
   864→  kelas: string,
   865→  n: number,
   866→  kesulitan: Kesulitan
   867→): GenSoal[] {
   868→  const topik = (TOPIK_GENERATOR[mapel]?.[jenjang]?.[kelas] ?? []).find((t) => t.id === topikId);
   869→  if (!topik) return [];
   870→  const out: GenSoal[] = [];
   871→  const seen = new Set<string>();
   872→  let guard = 0;
   873→  while (out.length < n && guard < n * 40) {
   874→    guard++;
   875→    const s = topik.gen(kesulitan, Math.random);
   876→    const key = s.q;
   877→    if (seen.has(key)) continue;
   878→    seen.add(key);
   879→    out.push(s);
   880→  }
   881→  return out;
   882→}

   884→/** label topik dari id (untuk arsip); dicari di semua bank generator */
   885→export function topikLabel(mapel: GenMapel, topikId: string, jenjang: string, kelas: string): string {
   886→  return (
   887→    TOPIK_GENERATOR[mapel]?.[jenjang]?.[kelas]?.find((t) => t.id === topikId)?.label ??
   888→    Object.values(TOPIK_GENERATOR)
   889→      .flatMap((bank) => Object.values(bank).flatMap((k) => Object.values(k)))
   890→      .flat()
   891→      .find((t) => t.id === topikId)?.label ??
   892→    topikId
   893→  );
   894→}
