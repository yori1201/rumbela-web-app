   256→export function SoalView() {
   257→  const { db, mutate } = useApp();
   258→  const peng = db.pengaturan;
   259→
   260→  const [mode, setMode] = useState<"otomatis" | "manual">("otomatis");
   261→  const [genMapel, setGenMapel] = useState<GenMapel>("matematika");
   262→  const [mapel, setMapel] = useState<string>("Matematika");
   263→  const [jenjang, setJenjang] = useState<string>("SD");
   264→  const [kelas, setKelas] = useState<string>("1");
   265→  const [topikId, setTopikId] = useState<string>(
   266→    TOPIK_GENERATOR.matematika.SD?.["1"]?.[0]?.id ?? ""
   267→  );
   268→  const [topikText, setTopikText] = useState<string>("");
   269→  const [jumlah, setJumlah] = useState<number>(10);
   270→  const [kesulitan, setKesulitan] = useState<Kesulitan>(2);
   271→  const [petunjuk, setPetunjuk] = useState<string>("Kerjakan seluruh soal di bawah ini dengan teliti!");
   272→  const [kunci, setKunci] = useState<boolean>(true);
   273→
   274→  const [manualQs, setManualQs] = useState<WorksheetQ[]>([blankQ(), blankQ(), blankQ(), blankQ(), blankQ()]);
   275→  const [sheet, setSheet] = useState<Worksheet | null>(null);
   276→  const [busy, setBusy] = useState(false);
   277→
   278→  const topikList = useMemo(
   279→    () => TOPIK_GENERATOR[genMapel]?.[jenjang]?.[kelas] ?? [],
   280→    [genMapel, jenjang, kelas]
   281→  );
   282→  const kelasOpsi = useMemo(
   283→    () => (mode === "otomatis" ? kelasTersedia(genMapel, jenjang) : KELAS_BY_JENJANG[jenjang] ?? []),
   284→    [mode, genMapel, jenjang]
   285→  );
   286→
   287→  const gantiGenMapel = (m: GenMapel) => {
   288→    setGenMapel(m);
   289→    const def = GEN_MAPEL.find((g) => g.id === m);
   290→    const j = def?.jenjang[0] ?? "SD";
   291→    const kk = kelasTersedia(m, j)[0] ?? "1";
   292→    setJenjang(j);
   293→    setKelas(kk);
   294→    setTopikId(TOPIK_GENERATOR[m]?.[j]?.[kk]?.[0]?.id ?? "");
   295→  };
   296→
   297→  const gantiJenjangKelas = (j: string, k?: string) => {
   298→    setJenjang(j);
   299→    const opsi = mode === "otomatis" ? kelasTersedia(genMapel, j) : KELAS_BY_JENJANG[j] ?? [];
   300→    const kk = k && opsi.includes(k) ? k : opsi[0] || "1";
   301→    setKelas(kk);
   302→    setTopikId(TOPIK_GENERATOR[genMapel]?.[j]?.[kk]?.[0]?.id ?? "");
   303→  };
   304→
   305→  const buat = () => {
   306→    if (mode === "otomatis") {
   307→      const gen = generateQuestions(genMapel, topikId, jenjang, kelas, jumlah, kesulitan);
   308→      if (!gen.length) {
   309→        toast({ title: "Topik belum tersedia", description: "Pilih topik generator untuk jenjang & kelas ini." });
   310→        return;
   311→      }
   312→      setSheet({
   313→        id: "", createdAt: todayStr(), mapel: GEN_MAPEL.find((g) => g.id === genMapel)?.label ?? "Matematika", jenjang, kelas,
   314→        topik: topikLabel(genMapel, topikId, jenjang, kelas), petunjuk, kunci,
   315→        questions: gen.map((g) => ({ id: uid(), text: g.q, kind: "isian" as SoalKind, answer: g.a })),
   316→      });
   317→    } else {
   318→      const qs = manualQs
   319→        .filter((q) => q.text.trim())
   320→        .map((q) => ({
   321→          ...q,
   322→          text: q.text.trim(),
   323→          answer: q.kind === "pg" ? q.answer.toLowerCase() : q.answer.trim(),
   324→          options: q.kind === "pg" ? (q.options ?? []).map((o) => o.trim()) : undefined,
   325→        }));
   326→      if (!qs.length) {
   327→        toast({ title: "Belum ada soal", description: "Tulis minimal satu soal pada mode Tulis sendiri." });
   328→        return;
   329→      }
   330→      setSheet({
   331→        id: "", createdAt: todayStr(), mapel, jenjang, kelas,
   332→        topik: topikText.trim() || `Latihan ${mapel}`, petunjuk, kunci, questions: qs,
   333→      });
   334→    }
   335→    toast({ title: "Worksheet siap", description: "Periksa pratinjau, lalu cetak atau simpan." });
   336→  };
   337→
   338→  const simpanArsip = () => {
   339→    if (!sheet) return;
   340→    const rec: Worksheet = { ...sheet, id: uid(), createdAt: sheet.createdAt || todayStr() };
   341→    mutate((d) => {
   342→      d.worksheets.unshift(rec);
   343→    });
   344→    setSheet(rec);
   345→    toast({ title: "Tersimpan di arsip", description: `${rec.mapel} · ${rec.topik}` });
   346→  };
   347→
   348→  const bukaArsip = (ws: Worksheet) => {
   349→    setSheet(ws);
   350→    setMapel(ws.mapel || "Matematika");
   351→    if (KELAS_BY_JENJANG[ws.jenjang]) {
   352→      setJenjang(ws.jenjang);
   353→      setKelas(ws.kelas || KELAS_BY_JENJANG[ws.jenjang][0]);
   354→    }
   355→    setTopikText(ws.topik);
   356→    setPetunjuk(ws.petunjuk);
   357→    setKunci(ws.kunci);
   358→    setManualQs(ws.questions.length ? ws.questions.map((q) => ({ ...q })) : [blankQ()]);
   359→    setMode("manual");
   360→    toast({ title: "Worksheet dibuka dari arsip", description: ws.topik });
   361→  };
   362→
   363→  const unduhPng = async (jenis: "soal" | "kunci") => {
   364→    if (!sheet) return;
   365→    setBusy(true);
   366→    try {
   367→      const canvas = await renderWorksheetCanvas(sheet, peng, jenis);
   368→      const blob = await canvasToBlob(canvas);
   369→      const a = document.createElement("a");
   370→      a.href = URL.createObjectURL(blob);
   371→      a.download = `rumbela-${slugify(sheet.topik)}-${jenis}.png`;
   372→      a.click();
   373→      setTimeout(() => URL.revokeObjectURL(a.href), 3000);
   374→    } finally {
   375→      setBusy(false);
   376→    }
   377→  };
   378→
   379→  const cetak = () => {
   380→    if (!sheet) return;
   381→    window.print();
   382→  };
   383→
   384→  return (
   385→    <div className="space-y-5">
   386→      <div className="grid gap-5 xl:grid-cols-[400px,1fr]">
   387→        {/* ============ FORM ============ */}
   388→        <div className="space-y-4">
   389→          <SectionCard
   390→            title="Susun worksheet"
   391→            desc="Penggunaan internal Rumbela — lembar latihan bergambar khusus"
   392→          >
   393→            {/* mode */}
   394→            <div className="grid grid-cols-2 gap-2">
   395→              <button
   396→                type="button"
   397→                onClick={() => setMode("otomatis")}
   398→                className={cn(
   399→                  "rounded-xl border-2 px-3 py-2.5 text-left text-sm font-bold transition-colors",
   400→                  mode === "otomatis" ? "border-sky-400 bg-sky-50 text-sky-700" : "border-slate-200 text-slate-500 hover:border-sky-200"
   401→                )}
   402→              >
   403→                <Wand2 className="mb-1 h-4 w-4" /> Otomatis
   404→                <span className="block text-[11px] font-semibold opacity-75">4 bank generator mapel</span>
   405→              </button>
   406→              <button
   407→                type="button"
   408→                onClick={() => setMode("manual")}
   409→                className={cn(
   410→                  "rounded-xl border-2 px-3 py-2.5 text-left text-sm font-bold transition-colors",
   411→                  mode === "manual" ? "border-emerald-400 bg-emerald-50 text-emerald-700" : "border-slate-200 text-slate-500 hover:border-emerald-200"
   412→                )}
