     7→     7→export interface GenSoal {
     8→     8→  q: string;
     9→     9→  a: string;
    10→    10→}

    12→    12→export interface TopikDef {
    13→    13→  id: string;

    17→    17→
    18→    18→/** bilangan bulat acak a..s.d. b (inklusif) */
    19→    19→const ri = (R: () => number, a: number, b: number): number =>
    20→    20→  a + Math.floor(R() * (b - a + 1));
    21→    21→const pick = <T,>(R: () => number, arr: T[]): T => arr[Math.floor(R() * arr.length)];
    22→    22→/** skala rentang berdasarkan tingkat kesulitan: 1=mudah 2=sedang 3=sulit */
    23→    23→const S = (lv: number, base: number, inc: number) => Math.round(base * (1 + (lv - 1) * inc));
    24→    24→
    25→    25→const GCD = (a: number, b: number): number => (b === 0 ? Math.abs(a) : GCD(b, a % b));
    26→    26→const LCM = (a: number, b: number): number => (a * b) / GCD(a, b);
    27→    27→const simp = (p: number, q: number): string => {
    28→    28→  const g = GCD(p, q) || 1;
    29→    29→  let pp = p / g, qq = q / g;
    30→    30→  if (qq < 0) { pp = -pp; qq = -qq; }
    31→    31→  return qq === 1 ? `${pp}` : `${pp}/${qq}`;
    32→    32→};
    33→    33→/** desimal gaya Indonesia: 3,5 */
    34→    34→const idn = (n: number): string => n.toString().replace(".", ",");

   859→   859→/** Buat n soal unik dari satu topik pada mapel generator tertentu. */
   860→   860→export function generateQuestions(
   861→   861→  mapel: GenMapel,
   862→   862→  topikId: string,
   863→   863→  jenjang: string,
   864→   864→  kelas: string,
   865→   865→  n: number,
   866→   866→  kesulitan: Kesulitan
   867→   867→): GenSoal[] {
   868→   868→  const topik = (TOPIK_GENERATOR[mapel]?.[jenjang]?.[kelas] ?? []).find((t) => t.id === topikId);
   869→   869→  if (!topik) return [];
   870→   870→  const out: GenSoal[] = [];
   871→   871→  const seen = new Set<string>();
   872→   872→  let guard = 0;
   873→   873→  while (out.length < n && guard < n * 40) {
   874→   874→    guard++;
   875→   875→    const s = topik.gen(kesulitan, Math.random);
   876→   876→    const key = s.q;
   877→   877→    if (seen.has(key)) continue;
   878→   878→    seen.add(key);
   879→   879→    out.push(s);
   880→   880→  }
   881→   881→  return out;
   882→   882→}

   891→   891→      .find((t) => t.id === topikId)?.label ??
   892→   892→    topikId
   893→   893→  );
   894→   894→}
