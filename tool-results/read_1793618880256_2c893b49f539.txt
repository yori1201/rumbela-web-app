    42→  return { id: uid(), text: "", kind, answer: "", ...(kind === "pg" ? { options: ["", "", "", ""] } : {}) };
    43→}

   147→
   148→export function SoalView() {
   149→  const { db, mutate } = useApp();
   150→  const peng = db.pengaturan;
   151→
   152→  const [mode, setMode] = useState<"otomatis" | "manual">("otomatis");
   153→  const [genMapel, setGenMapel] = useState<GenMapel>("matematika");
   154→  const [mapel, setMapel] = useState<string>("Matematika");
   155→  const [jenjang, setJenjang] = useState<string>("SD");
   156→  const [kelas, setKelas] = useState<string>("1");
   157→  const [topikId, setTopikId] = useState<string>(
   158→    TOPIK_GENERATOR.matematika.SD?.["1"]?.[0]?.id ?? ""
   159→  );
   160→  const [topikText, setTopikText] = useState<string>("");
   161→  const [jumlah, setJumlah] = useState<number>(10);
   162→  const [kesulitan, setKesulitan] = useState<Kesulitan>(2);
   163→  const [petunjuk, setPetunjuk] = useState<string>("Kerjakan seluruh soal di bawah ini dengan teliti!");
   164→  const [kunci, setKunci] = useState<boolean>(true);
   165→
   166→  const [manualQs, setManualQs] = useState<WorksheetQ[]>([blankQ(), blankQ(), blankQ(), blankQ(), blankQ()]);
   167→  const [sheet, setSheet] = useState<Worksheet | null>(null);
   168→  const [busy, setBusy] = useState(false);
   169→
   170→  const topikList = useMemo(
   171→    () => TOPIK_GENERATOR[genMapel]?.[jenjang]?.[kelas] ?? [],
   172→    [genMapel, jenjang, kelas]
   173→  );
   174→  const kelasOpsi = useMemo(
   175→    () => (mode === "otomatis" ? kelasTersedia(genMapel, jenjang) : KELAS_BY_JENJANG[jenjang] ?? []),
   176→    [mode, genMapel, jenjang]
   177→  );
   178→
   179→  const gantiGenMapel = (m: GenMapel) => {
   180→    setGenMapel(m);
   181→    const def = GEN_MAPEL.find((g) => g.id === m);
   182→    const j = def?.jenjang[0] ?? "SD";
   183→    const kk = kelasTersedia(m, j)[0] ?? "1";
   184→    setJenjang(j);
   185→    setKelas(kk);
   186→    setTopikId(TOPIK_GENERATOR[m]?.[j]?.[kk]?.[0]?.id ?? "");
   187→  };
   188→
   189→  const gantiJenjangKelas = (j: string, k?: string) => {
   190→    setJenjang(j);
   191→    const opsi = mode === "otomatis" ? kelasTersedia(genMapel, j) : KELAS_BY_JENJANG[j] ?? [];
   192→    const kk = k && opsi.includes(k) ? k : opsi[0] || "1";
   193→    setKelas(kk);
   194→    setTopikId(TOPIK_GENERATOR[genMapel]?.[j]?.[kk]?.[0]?.id ?? "");
   195→  };
   196→
   197→  const buat = () => {
   198→    if (mode === "otomatis") {
   199→      const gen = generateQuestions(genMapel, topikId, jenjang, kelas, jumlah, kesulitan);
   200→      if (!gen.length) {
   201→        toast({ title: "Topik belum tersedia", description: "Pilih topik generator untuk jenjang & kelas ini." });
   202→        return;
   203→      }
   204→      setSheet({
   205→        id: "", createdAt: todayStr(), mapel: GEN_MAPEL.find((g) => g.id === genMapel)?.label ?? "Matematika", jenjang, kelas,
   206→        topik: topikLabel(genMapel, topikId, jenjang, kelas), petunjuk, kunci,
   207→        questions: gen.map((g) => ({ id: uid(), text: g.q, kind: "isian" as SoalKind, answer: g.a })),
   208→      });
   209→    } else {
   210→      const qs = manualQs
   211→        .filter((q) => q.text.trim())
   212→        .map((q) => ({
   213→          ...q,
   214→          text: q.text.trim(),
   215→          answer: q.kind === "pg" ? q.answer.toLowerCase() : q.answer.trim(),
   216→          options: q.kind === "pg" ? (q.options ?? []).map((o) => o.trim()) : undefined,
   217→        }));
   218→      if (!qs.length) {
   219→        toast({ title: "Belum ada soal", description: "Tulis minimal satu soal pada mode Tulis sendiri." });
   220→        return;
   221→      }
   222→      setSheet({
   223→        id: "", createdAt: todayStr(), mapel, jenjang, kelas,
   224→        topik: topikText.trim() || `Latihan ${mapel}`, petunjuk, kunci, questions: qs,
   225→      });
   226→    }
   227→    toast({ title: "Worksheet siap", description: "Periksa pratinjau, lalu cetak atau simpan." });
   228→  };
   229→
   230→  const simpanArsip = () => {
   231→    if (!sheet) return;
   232→    const rec: Worksheet = { ...sheet, id: uid(), createdAt: sheet.createdAt || todayStr() };
   233→    mutate((d) => {
   234→      d.worksheets.unshift(rec);
   235→    });
   236→    setSheet(rec);
   237→    toast({ title: "Tersimpan di arsip", description: `${rec.mapel} · ${rec.topik}` });
   238→  };
   239→
   240→  const bukaArsip = (ws: Worksheet) => {
   241→    setSheet(ws);
   242→    setMapel(ws.mapel || "Matematika");
   243→    if (KELAS_BY_JENJANG[ws.jenjang]) {
   244→      setJenjang(ws.jenjang);
   245→      setKelas(ws.kelas || KELAS_BY_JENJANG[ws.jenjang][0]);
   246→    }
   247→    setTopikText(ws.topik);
   248→    setPetunjuk(ws.petunjuk);
   249→    setKunci(ws.kunci);
   250→    setManualQs(ws.questions.length ? ws.questions.map((q) => ({ ...q })) : [blankQ()]);
   251→    setMode("manual");
   252→    toast({ title: "Worksheet dibuka dari arsip", description: ws.topik });
   253→  };
   254→
   255→  const unduhPng = async (jenis: "soal" | "kunci") => {
   256→    if (!sheet) return;
   257→    setBusy(true);
   258→    try {
   259→      const canvas = await renderWorksheetCanvas(sheet, peng, jenis);
   260→      const blob = await canvasToBlob(canvas);
   261→      const a = document.createElement("a");
   262→      a.href = URL.createObjectURL(blob);
   263→      a.download = `rumbela-${slugify(sheet.topik)}-${jenis}.png`;
   264→      a.click();
   265→      setTimeout(() => URL.revokeObjectURL(a.href), 3000);
   266→    } finally {
   267→      setBusy(false);
   268→    }
   269→  };
   270→
   271→  const cetak = () => {
   272→    if (!sheet) return;
   273→    window.print();
   274→  };
   275→
   276→  return (
   277→    <div className="space-y-5">
   278→      <div className="grid gap-5 xl:grid-cols-[400px,1fr]">
   279→        {/* ============ FORM ============ */}
   280→        <div className="space-y-4">
   281→          <SectionCard
   282→            title="Susun worksheet"
   283→            desc="Penggunaan internal Rumbela — lembar latihan bergambar khusus"
   284→          >
   285→            {/* mode */}
   286→            <div className="grid grid-cols-2 gap-2">
   287→              <button
   288→                type="button"
   289→                onClick={() => setMode("otomatis")}
   290→                className={cn(
   291→                  "rounded-xl border-2 px-3 py-2.5 text-left text-sm font-bold transition-colors",
   292→                  mode === "otomatis" ? "border-sky-400 bg-sky-50 text-sky-700" : "border-slate-200 text-slate-500 hover:border-sky-200"
   293→                )}
   294→              >
   295→                <Wand2 className="mb-1 h-4 w-4" /> Otomatis
   296→                <span className="block text-[11px] font-semibold opacity-75">4 bank generator mapel</span>
   297→              </button>
   298→              <button
   299→                type="button"
   300→                onClick={() => setMode("manual")}
   301→                className={cn(
   302→                  "rounded-xl border-2 px-3 py-2.5 text-left text-sm font-bold transition-colors",
   303→                  mode === "manual" ? "border-emerald-400 bg-emerald-50 text-emerald-700" : "border-slate-200 text-slate-500 hover:border-emerald-200"
   304→                )}
   305→              >
   306→                <BookPlus className="mb-1 h-4 w-4" /> Tulis sendiri
   307→                <span className="block text-[11px] font-semibold opacity-75">Semua mata pelajaran</span>
   308→              </button>
   309→            </div>
   310→
   311→            {/* mapel */}
   312→            <div className="mt-4 space-y-1.5">
   313→              <Label>Mata pelajaran</Label>
   314→              {mode === "otomatis" ? (
   315→                <Select value={genMapel} onValueChange={(v) => gantiGenMapel(v as GenMapel)}>
   316→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   317→                  <SelectContent>
   318→                    {GEN_MAPEL.map((g) => (
   319→                      <SelectItem key={g.id} value={g.id}>{g.label} — {g.rentang}</SelectItem>
   320→                    ))}
   321→                  </SelectContent>
   322→                </Select>
   323→              ) : (
   324→                <Select value={mapel} onValueChange={setMapel}>
   325→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   326→                  <SelectContent>
   327→                    {MAPEL_LIST.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   328→                  </SelectContent>
   329→                </Select>
   330→              )}
   331→            </div>
   332→
   333→            {/* jenjang & kelas */}
   334→            <div className="mt-3 grid grid-cols-2 gap-3">
   335→              <div className="space-y-1.5">
   336→                <Label>Jenjang</Label>
   337→                <div className="flex overflow-hidden rounded-lg border border-slate-200">
   338→                  {JENJANG.map((j) => {
   339→                    const tersedia = mode === "manual" || (GEN_MAPEL.find((g) => g.id === genMapel)?.jenjang ?? []).includes(j);
   340→                    return (
   341→                      <button
   342→                        key={j}
   343→                        type="button"
   344→                        disabled={!tersedia}
   345→                        onClick={() => gantiJenjangKelas(j)}
   346→                        className={cn(
   347→                          "flex-1 py-2 text-sm font-bold transition-colors",
   348→                          jenjang === j ? "bg-sky-600 text-white" : tersedia ? "bg-white text-slate-500 hover:bg-sky-50" : "cursor-not-allowed bg-slate-100 text-slate-300"
   349→                        )}
   350→                      >
   351→                        {j}
   352→                      </button>
   353→                    );
   354→                  })}
   355→                </div>
   356→              </div>
   357→              <div className="space-y-1.5">
   358→                <Label>Kelas</Label>
   359→                <Select value={kelas} onValueChange={(k) => gantiJenjangKelas(jenjang, k)}>
   360→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   361→                  <SelectContent>
   362→                    {kelasOpsi.map((k) => (
   363→                      <SelectItem key={k} value={k}>Kelas {k}</SelectItem>
   364→                    ))}
   365→                  </SelectContent>
   366→                </Select>
   367→              </div>
   368→            </div>
   369→
   370→            {/* topik */}
   371→            <div className="mt-3 space-y-1.5">
   372→              <Label>Topik soal</Label>
   373→              {mode === "otomatis" ? (
   374→                <Select value={topikId} onValueChange={setTopikId}>
   375→                  <SelectTrigger className="w-full"><SelectValue placeholder="Pilih topik" /></SelectTrigger>
   376→                  <SelectContent>
   377→                    {topikList.map((t) => (
   378→                      <SelectItem key={t.id} value={t.id}>{t.label}</SelectItem>
   379→                    ))}
   380→                  </SelectContent>
   381→                </Select>
   382→              ) : (
   383→                <Input
   384→                  value={topikText}
   385→                  onChange={(e) => setTopikText(e.target.value)}
   386→                  placeholder="cth. Teks Deskripsi, Perkalian Pecahan…"
   387→                />
   388→              )}
   389→            </div>
   390→
   391→            {/* jumlah & kesulitan (otomatis) */}
   392→            {mode === "otomatis" && (
   393→              <div className="mt-3 grid grid-cols-2 gap-3">
   394→                <div className="space-y-1.5">
   395→                  <Label>Jumlah soal</Label>
   396→                  <Select value={String(jumlah)} onValueChange={(v) => setJumlah(Number(v))}>
   397→                    <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   398→                    <SelectContent>
   399→                      {JUMLAH_OPSI.map((n) => <SelectItem key={n} value={String(n)}>{n} soal</SelectItem>)}
   400→                    </SelectContent>
   401→                  </Select>
   402→                </div>
   403→                <div className="space-y-1.5">
   404→                  <Label>Kesulitan</Label>
   405→                  <div className="flex overflow-hidden rounded-lg border border-slate-200">
   406→                    {([["1", "Mudah"], ["2", "Sedang"], ["3", "Sulit"]] as const).map(([v, l]) => (
   407→                      <button
   408→                        key={v}
   409→                        type="button"
   410→                        onClick={() => setKesulitan(Number(v) as Kesulitan)}
   411→                        className={cn(
   412→                          "flex-1 py-2 text-xs font-bold transition-colors",
   413→                          kesulitan === Number(v) ? "bg-emerald-500 text-white" : "bg-white text-slate-500 hover:bg-emerald-50"
   414→                        )}
   415→                      >
   416→                        {l}
   417→                      </button>
   418→                    ))}
   419→                  </div>
   420→                </div>
   421→              </div>
   422→            )}
   423→
   424→            {/* petunjuk */}
   425→            <div className="mt-3 space-y-1.5">
   426→              <Label>Petunjuk pengerjaan (opsional)</Label>
   427→              <Textarea
   428→                value={petunjuk}
   429→                onChange={(e) => setPetunjuk(e.target.value)}
   430→                rows={2}
   431→                placeholder="cth. Kerjakan pada lembar ini, tunjukkan cara pengerjaanmu!"
   432→              />
   433→            </div>
   434→
   435→            {/* kunci */}
   436→            <div className="mt-3 flex items-center justify-between rounded-xl border border-emerald-100 bg-emerald-50/60 px-4 py-3">
   437→              <div>
   438→                <p className="text-sm font-bold text-emerald-800">Lembar kunci jawaban</p>
   439→                <p className="text-xs text-emerald-700/80">Ikut tercetak & bisa diunduh terpisah</p>
   440→              </div>
   441→              <Switch checked={kunci} onCheckedChange={setKunci} />
   442→            </div>
   443→
